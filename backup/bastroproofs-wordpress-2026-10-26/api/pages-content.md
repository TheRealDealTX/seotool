(() => {
  const setAnswerHeight = (item) => {
    const answer = item.querySelector('.faq-answer');
    if (answer) item.style.setProperty('--answer-height', `${answer.scrollHeight}px`);
  };

  const prepareFaqs = (root = document) => {
    root.querySelectorAll('.page .faq-item').forEach((item) => {
      setAnswerHeight(item);
      const button = item.querySelector('.faq-question');
      if (button) button.setAttribute('aria-expanded', String(item.classList.contains('open')));
    });
  };

  document.addEventListener('click', (event) => {
    const button = event.target.closest('.page .faq-question');
    if (!button) return;
    const item = button.closest('.faq-item');
    if (!item) return;
    setAnswerHeight(item);
    item.classList.toggle('open');
    button.setAttribute('aria-expanded', String(item.classList.contains('open')));
  });

  document.addEventListener('submit', (event) => {
    const form = event.target.closest('.page [data-estimate-form]');
    if (!form) return;
    event.preventDefault();
    window.alert('Thanks! Connect this form to your preferred form handler before launch.');
  });

  const start = () => {
    prepareFaqs();
    const observer = new MutationObserver((mutations) => {
      mutations.forEach((mutation) => {
        mutation.addedNodes.forEach((node) => {
          if (node.nodeType === 1) prepareFaqs(node);
        });
      });
    });
    observer.observe(document.body, { childList: true, subtree: true });
    window.addEventListener('resize', () => prepareFaqs(), { passive: true });
  };

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', start, { once: true });
  } else {
    start();
  }
})();
